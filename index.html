Webform1.aspx.cs

using System;
namespace WebControlsDemo
{
    public partial class Default : System.Web.UI.Page
    {
        protected void btnSubmit_Click(object sender, EventArgs e)
        {
            string name = txtName.Text;
            string age = txtAge.Text;
            string gender = "";
            if (rbMale.Checked)
              gender = "Male";
            else if (rbFemale.Checked)
              gender = "Female";
            string course = ddlCourse.SelectedItem.Text;
            string city = lstCity.SelectedItem.Text;
            string skills = "";
            if (chkJava.Checked)
              skills += "Java ";
            if (chkPython.Checked)
              skills += "Python ";
            if (chkCSharp.Checked)
              skills += "C# ";
            lblResult.Text =
              "Name: " + name + "<br/>" +
              "Age: " + age + "<br/>" +
              "Gender: " + gender + "<br/>" +
              "Course: " + course + "<br/>" +
              "Skills: " + skills + "<br/>" +
              "City: " + city;
        }
    }
}

EX 4 – Cookies

Webform1.aspx.cs

using System;
using System.Web;
public partial class Cookies : System.Web.UI.Page
{
    protected void Page_Load(object sender, EventArgs e)
    {
    }
    protected void InsertCookie()
    {
        HttpCookie cookie = new HttpCookie("Student");
        cookie["Name"] = "Leo";
        cookie["Course"] = "BCA";
        cookie.Expires = DateTime.Now.AddDays(7);

        Response.Cookies.Add(cookie);
    }
    protected void ReadCookie()
    {
        HttpCookie cookie = Request.Cookies["Student"];

        if (cookie != null)
        {
            string name = cookie["Name"];
            string course = cookie["Course"];
            Response.Write("Name: " + name + "<br>");
            Response.Write("Course: " + course);
        }
    }
    protected void UpdateCookie()
    {
        HttpCookie cookie = Request.Cookies["Student"];
        if (cookie != null)
        {
            cookie["Name"] = "Joel";
            cookie["Course"] = "B.Sc Computer Science";
            cookie.Expires = DateTime.Now.AddDays(7);

            Response.Cookies.Set(cookie);
        }
    }
    protected void DeleteCookie()
    {
        HttpCookie cookie = Request.Cookies["Student"];
        if (cookie != null)
        {
            cookie.Expires = DateTime.Now.AddDays(-1);
            Response.Cookies.Add(cookie);
        }
    }
}

EX 5 – ASP.NET State Management

Webform1.aspx.cs

using System;

public partial class StateManagement : System.Web.UI.Page
{
    protected void Page_Load(object sender, EventArgs e)
    {
    }
    protected void btnSet_Click(object sender, EventArgs e)
    {
        Session["Name"] = "Leo";
        Session["Course"] = "BCA";

        Response.Write("Session values stored successfully.");
    }
    protected void btnGet_Click(object sender, EventArgs e)
    {
        if (Session["Name"] != null)
        {
            Response.Write("Name: " + Session["Name"] + "<br>");
            Response.Write("Course: " + Session["Course"]);
        }
        else
        {
            Response.Write("Session is empty.");
        }
    }
    protected void btnUpdate_Click(object sender, EventArgs e)
    {
        Session["Name"] = "Joel";
        Session["Course"] = "B.Sc Computer Science";

        Response.Write("Session values updated successfully.");
    }
    protected void btnDelete_Click(object sender, EventArgs e)
    {
        Session.Remove("Name");
        Session.Remove("Course");
        Response.Write("Session values removed successfully.");
    }
}

EX 6 – Database Operations

Webform1.aspx.cs

using System;
using System.Data;
using System.Data.SqlClient;
public partial class Database : System.Web.UI.Page
{
    string connectionString =
        @"Data Source=.;Initial Catalog=CollegeDB;Integrated Security=True";
    protected void InsertData()
    {
        using (SqlConnection con = new SqlConnection(connectionString))
        {
            string query = "INSERT INTO Student(Name, Course, Mark) " +
                    "VALUES(@Name, @Course, @Mark)";
            SqlCommand cmd = new SqlCommand(query, con);
            cmd.Parameters.AddWithValue("@Name", "Leo");
            cmd.Parameters.AddWithValue("@Course", "BCA");
            cmd.Parameters.AddWithValue("@Mark", 85);
            con.Open();
            cmd.ExecuteNonQuery();

            Response.Write("Record Inserted Successfully");
        }
    }
    protected void UpdateData()
    {
        using (SqlConnection con = new SqlConnection(connectionString))
        {
            string query = "UPDATE Student SET Name=@Name, " +
                    "Course=@Course, Mark=@Mark WHERE Id=@Id";
            SqlCommand cmd = new SqlCommand(query, con);
            cmd.Parameters.AddWithValue("@Id", 1);
            cmd.Parameters.AddWithValue("@Name", "Joel");
            cmd.Parameters.AddWithValue("@Course", "BCA");
            cmd.Parameters.AddWithValue("@Mark", 90);
            con.Open();
            cmd.ExecuteNonQuery();
            Response.Write("Record Updated Successfully");
        }
    }
    protected void DeleteData()
    {
        using (SqlConnection con = new SqlConnection(connectionString))
        {
            string query = "DELETE FROM Student WHERE Id=@Id";
            SqlCommand cmd = new SqlCommand(query, con);
            cmd.Parameters.AddWithValue("@Id", 1);
            con.Open();
            cmd.ExecuteNonQuery();
            Response.Write("Record Deleted Successfully");
        }
    }
    protected void ReadData()
    {
        using (SqlConnection con = new SqlConnection(connectionString))
        {
            string query = "SELECT * FROM Student";
            SqlDataAdapter da = new SqlDataAdapter(query, con);
            DataTable dt = new DataTable();
            da.Fill(dt);
            foreach (DataRow row in dt.Rows)
            {
                Response.Write(
                     "ID: " + row["Id"] +
                     " Name: " + row["Name"] +
                     " Course: " + row["Course"] +
                     " Mark: " + row["Mark"] +
                     "<br>"
                );
            }
        }
    }
}
